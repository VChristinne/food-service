\section{Gestão de Clientes}

\subsection{POST /api/v1/customers}

\noindent\textbf{Finalidade}: Criar um novo cliente

\noindent\textbf{Permissão}: Público

\noindent\textbf{Body}: Todos os campos são obrigatórios exceto \texttt{complement}

\vspace{-0.2cm}

\begin{minted}{json}
{
  "name": "{{ _.firstName }} {{ _.lastName }}",
  "password": "12345678!C",
  "email": "{{ _.firstName }}{{ _.lastName }}@outlook.com",
  "phone": "{% faker 'randomPhoneNumber' %}",
  "cep": "22775055",
  "complement": "AP 10"
}
\end{minted}

\subsection{PATCH /api/v1/customers/:customer\_id}

\noindent\textbf{Finalidade}: Atualizar informações de um cliente específico

\noindent\textbf{Permissão}: JWT | Perfil: Cliente

\noindent\textbf{Body}: Todos os campos são opcionais

\begin{minted}{json}
{
  "name": "{{ _.firstName }} {{ _.lastName }}",
  "email": "{{ _.firstName }}{{ _.lastName }}@gmail.com"
}
\end{minted}

\noindent\textbf{Status Codes}: 200, 401, 403, 404, 422

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{200 OK}]
Para fins de visualização breve, é mostrado apenas os campos atualizados.
\begin{minted}{json}
{
  "name": "Reed Schoen",
  "email": "ReedSchoen@gmail.com",
  "updated_at": 1778247720,
  "id": "069fdd1e-30e8-7145-8000-622d09ef189d",
}
\end{minted}
\end{tcolorbox}
