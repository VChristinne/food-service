\section{Gestão de Funcionários}

\subsection{POST /api/v1/employees}

\noindent\textbf{Finalidade}: Criar um novo funcionário

\noindent\textbf{Permissão}: JWT | Perfil: Admin

\noindent\textbf{Body}:
Todos os campos são obrigatórios exceto \texttt{complement}

\vspace{-0.2cm}

\begin{minted}{json}
{
  "name": "{{ _.firstName }} {{ _.lastName }}",
  "password": "12345678!Ch",
  "email": "{{ _.firstName | lower }}.{{ _.lastName | lower }}@{{ _.companyEmail }}",
  "phone": "{% faker 'randomPhoneNumber' %}",
  "cep": "22775046",
  "complement": "AP 10",
  "store_id": "{{ _.storeID }}",
  "role": "chef"
}
\end{minted}

\noindent\textbf{Permissão}: JWT | Perfis: Admin, Manager

\noindent\textbf{Body}: Nenhum campo é necessário

\subsection{PATCH /api/v1/employees/:employee\_id}

\noindent\textbf{Finalidade}: Atualizar detalhes de um funcionário específico

\noindent\textbf{Permissão}: JWT | Perfil: Admin, Manager (apenas para funcionários da mesma loja)

\noindent\textbf{Body}: Todos os campos são opcionais

\vspace{-0.2cm}

\begin{minted}{json}
{
  "name": "{{ _.firstName }} {{ _.lastName }}",
  "email": "{{ _.firstName | lower }}.{{ _.lastName | lower }}@{{ _.companyEmail }}",
  "phone": "{% faker 'randomPhoneNumber' %}"
}
\end{minted}

\noindent\textbf{Status Codes}: 200, 401, 403, 404, 422

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{200 OK}]
