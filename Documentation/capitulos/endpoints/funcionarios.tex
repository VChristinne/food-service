\section{Gestão de Funcionários}

\subsection{Listar Funcionários}

\begin{itemize}[topsep=1pt, itemsep=0pt]
  \item[]\textbf{Finalidade}: Listar todos os funcionários registrados.

  \item[]\textbf{Rota}: GET /api/v1/employees

  \item[]\textbf{Permissão}: JWT | Perfis: Admin, Manager

  \item[]\textbf{Parâmetros}:
  \begin{itemize}[topsep=1pt, itemsep=0pt]
    \item Path: Nenhum
    \item Query: page (mínimo 1, padrão 1), page\_size (mínimo 1, máximo 100, padrão 10)
  \end{itemize}

  \textbf{Request}:
  \vspace{-0.3cm}
  \begin{minted}{json}
  {}
  \end{minted}

  \textbf{Status codes}:
  \begin{itemize}[topsep=1pt, itemsep=0pt]
    \item 200 OK: Retorna uma lista paginada de funcionários.
    \item 401 Unauthorized: Token JWT ausente ou inválido.
    \item 403 Forbidden: Permissão insuficiente para acessar o recurso.
    \item 422 Unprocessable Entity: Parâmetros de consulta inválidos.
  \end{itemize}
