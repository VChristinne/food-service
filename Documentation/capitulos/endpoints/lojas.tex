\section{Gestão de Lojas}

\subsection{\texttt{POST /api/v1/stores}}

\noindent
\textbf{Finalidade}: Criar uma nova loja \\
\textbf{Permissão}: JWT | Perfil: Admin \\
\textbf{Body}\footnote{Para fins de teste, é usado a biblioteca Faker para gerar dados aleatórios.}:
Todos os campos são obrigatórios exceto \texttt{complement}

\vspace{-0.2cm}

\begin{minted}{json}
{
  "phone": "{% faker 'randomPhoneNumber' %}",
  "cep": "22775056",
  "complement": "10"
}
\end{minted}

\noindent\textbf{Status Codes}: 201, 403, 422

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{201 Created}]
\begin{minted}{json}
{
  "created_at": 1777558556,
  "phone": "(573) 327-4345 x43932",
  "deleted_at": null,
  "address": {
    "cep": "22775-056",
    "state": "RJ",
    "city": "Rio de Janeiro",
    "neighborhood": "Barra da Tijuca",
    "street": "Avenida José Silva de Azevedo Neto",
    "complement": "10"
  },
  "id": "069f3641-c09e-7731-8000-b04e75d39158",
  "updated_at": 1777558556
}
\end{minted}
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5, colframe=black, title=\textbf{403 Forbidden}]
Permissão insuficiente para acessar o recurso \\
\small{\textit{Para mais detalhes, consulte o \ref{itm:403-response} na \ref{sec:respostas-erro-padrao}}}
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5, colframe=black, title=\textbf{422 Unprocessable Entity}]
Formato de requisição inválido \\
\small{\textit{Para mais detalhes, consulte o \ref{itm:422-response} na \ref{sec:respostas-erro-padrao}}}
\end{tcolorbox}

\subsection{\texttt{GET /api/v1/stores}}

\noindent\textbf{Finalidade}: Listar todas as lojas

\noindent\textbf{Permissão}: JWT | Perfis: Admin

\noindent\textbf{Body}: Nenhum campo é necessário

\noindent\textbf{Query Parameters}\footnote{Para fins de documentação, a resposta é paginada com um item por página.}:
\begin{itemize}[topsep=1pt, itemsep=0pt]
  \item\texttt{page} (integer, opcional, mínimo 1, padrão 1)
  \item\texttt{page\_size} (integer, opcional, mínimo 1, máximo 100, padrão 10)
\end{itemize}

\noindent\textbf{Status Codes}: 200, 403, 422

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{200 OK}]
\begin{minted}{json}
{
  "data": [
    {
      "phone": "477-487-1816 x452",
      "created_at": 1777558516,
      "deleted_at": null,
      "updated_at": 1777558516,
      "id": "069f363f-4619-75bb-8000-db08e1b84c0e",
      "address": {
        "cep": "22775-058",
        "state": "RJ",
        "city": "Rio de Janeiro",
        "neighborhood": "Jacarepaguá",
        "street": "Rua Imperatriz Leopoldina",
        "complement": "103"
      }
    }
  ],
  "total": 3,
  "page": 1,
  "page_size": 1,
  "total_pages": 3
}
\end{minted}
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5, colframe=black, title=\textbf{403 Forbidden}]
Permissão insuficiente para acessar o recurso \\
\small{\textit{Para mais detalhes, consulte o \ref{itm:403-response} na \ref{sec:respostas-erro-padrao}}}
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5, colframe=black, title=\textbf{422 Unprocessable Entity}]
Numero de página ou tamanho de página inválido \\
\small{\textit{Para mais detalhes, consulte o \ref{itm:422-response} na \ref{sec:respostas-erro-padrao}}}
\end{tcolorbox}

\subsection{\texttt{GET /api/v1/stores/:store\_id}}

\noindent\textbf{Finalidade}: Obter detalhes de uma loja específica

\noindent\textbf{Permissão}: JWT | Perfis: Admin

\noindent\textbf{Body}: Nenhum campo é necessário

\noindent\textbf{Path Parameters}: \texttt{store\_id} (string, obrigatório)

\noindent\textbf{Status Codes}: 200, 403, 404

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{200 OK}]
\begin{minted}{json}

  "updated_at": 1777558556,

    "city": "Rio de Janeiro",
    "neighborhood": "Barra da Tijuca",

    "complement": "10"
  }
}
\end{minted}
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5, colframe=black, title=\textbf{403 Forbidden}]
