\section{Autenticação de Usuários}

\subsection{Login}

\begin{itemize}[topsep=1pt, itemsep=0pt]
  \item[]\textbf{Finalidade}: Autenticar usuários e retornar token de acesso.

  \item[]\textbf{Rota}: POST /api/v1/auth/login

  \item[]\textbf{Permissão}: Público

  \item[]\textbf{Parâmetros}:
  \begin{itemize}[topsep=1pt, itemsep=0pt]
    \item Path: Nenhum
    \item Query: Nenhum
  \end{itemize}

  \item[]\textbf{Request}:
  \vspace{-0.3cm}
  \begin{minted}{json}
  {
    "email": "Christinne@raizesnordeste.com",
    "password": "12345678!A"
  }
  \end{minted}

  \textbf{Status codes}:
  \begin{itemize}[topsep=1pt, itemsep=0pt]
    \item 200 OK: Login bem-sucedido, token de acesso retornado.
    \item 401 Unauthorized: Credenciais inválidas, acesso negado.
    \item 422 Unprocessable Content: Formato de requisição inválido.
  \end{itemize}

  \item[]\textbf{Response 200}\footnote{Token JWT encurtado para fins de documentação.}:
  \vspace{-0.3cm}
  \begin{minted}{json}
  {
    "id": "069d510e-a565-7add-8000-65391d41016c",
    "access_token": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
    "token_type": "Bearer"
  }
  \end{minted}

  \item[]\textbf{Erro 401}:
  \vspace{-0.3cm}
  \begin{minted}{json}
  { "detail": "Email or password is invalid" }
  \end{minted}

  \item[]\textbf{Erro 422}:
  \vspace{-0.3cm}
  \begin{minted}{json}
  {
    "detail": [
      {
        "type": "missing",
        "loc": [
          "body",
          "email"
        ],
        "msg": "Field required",
        "input": {
          "": "Christinne@raizesnordeste.com",
          "password": "12345678!A"
        }
      }
    ]
  }
  \end{minted}
\end{itemize}

\subsection{Logout}

\begin{itemize}[topsep=1pt, itemsep=0pt]
  \item[]\textbf{Finalidade}: Encerrar sessão do usuário.

  \item[]\textbf{Rota}: POST /api/v1/auth/logout

  \item[]\textbf{Permissão}: JWT

  \item[]\textbf{Parâmetros}:
  \begin{itemize}[topsep=1pt, itemsep=0pt]
    \item Path: Nenhum
    \item Query: Nenhum
  \end{itemize}

  \item[]\textbf{Request}:
  \vspace{-0.3cm}
  \begin{minted}{json}
  {}
  \end{minted}

  \textbf{Status codes}:
  \begin{itemize}[topsep=1pt, itemsep=0pt]
    \item 200 OK: Logout bem-sucedido\footnote{JWT é stateless, sendo necessário implementar blacklist para invalidar tokens.
    Para fins de documentação, considero o logout como uma operação de sucesso.}.
    \item 401 Unauthorized: Token de acesso inválido ou ausente.
  \end{itemize}

  \item[]\textbf{Response 200}:
  \vspace{-0.3cm}
  \begin{minted}{json}
  { "message": "Logout successful" }
  \end{minted}

  \item[]\textbf{Erro 401}:
  \vspace{-0.3cm}
  \begin{minted}{json}
  { "detail": "Invalid token" }
  \end{minted}

\end{itemize}
