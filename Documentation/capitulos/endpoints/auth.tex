\section{Autenticação de Usuários}

\subsection{\texttt{POST /api/v1/auth/login}}

\noindent\textbf{Finalidade}: Autenticar usuários

\noindent\textbf{Permissão}: Público

\noindent\textbf{Body}: Todos os campos são obrigatórios

\vspace{-0.2cm}

\begin{minted}{json}
{
  "email": "christinne@raizesnordeste.com",
  "password": "12345678!A"
}
\end{minted}

\noindent\textbf{Status Codes}: 200, 401, 422

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{200 OK}]
Para fins de documentação, o token é truncado na visualização.
\begin{minted}{json}
{
  "id": "069d510e-a565-7add-8000-65391d41016c",
  "access_token": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
  "token_type": "Bearer"
}
\end{minted}
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5, colframe=black, title=\textbf{401 Unauthorized}]
Credenciais inválidas \\
\small{\textit{Para mais detalhes, consulte o \ref{itm:401-response} na \ref{sec:respostas-erro-padrao}}}
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5, colframe=black, title=\textbf{422 Unprocessable Entity}]
Formato de requisição inválido \\
\small{\textit{Para mais detalhes, consulte o \ref{itm:422-response} na \ref{sec:respostas-erro-padrao}}}
\end{tcolorbox}

\subsection{\texttt{POST /api/v1/auth/logout}}

\noindent\textbf{Finalidade}: Encerrar sessão do usuário

\noindent\textbf{Permissão}: JWT | Perfis: Todos os usuários autenticados

\noindent\textbf{Body}: Nenhum campo é necessário

\noindent\textbf{Status Codes}: 200, 401

\begin{tcolorbox}[colback=green!10, colframe=black, title=\textbf{200 OK}]
Logout simulado, pois o JWT é stateless e não pode ser invalidado pelo servidor. \\
Todos os tokens expiram em 1 hora.

\begin{minted}{json}
{
  "message": "Logout successful"
}
\end{minted}
\end{tcolorbox}

\begin{tcolorbox}[colback=red!10, colframe=black, title=\textbf{401 Unauthorized}]
Token de autenticação inválido ou ausente \\
\small{\textit{Para mais detalhes, consulte o \ref{itm:401-response} na \ref{sec:respostas-erro-padrao}}}
\end{tcolorbox}
