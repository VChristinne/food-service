\section{Autenticação de Usuários}

\subsection{POST /api/v1/auth/login}

\fbox{\begin{minipage}{0.95\textwidth}
\textbf{Finalidade:} Autenticar usuários

\textbf{Permissão:} Público
\end{minipage}}

\vspace{0.5cm}

\noindent\textbf{Request Body}:

\vspace{-0.2cm}

\begin{minted}{text}
email      string, obrigatório
password   string, obrigatório
\end{minted}

\begin{minted}{json}
{
  "email": "christinne@raizesnordeste.com",
  "password": "12345678!A"
}
\end{minted}

\noindent\textbf{Status Codes}: 200, 401, 422

\vspace{0.4cm}

\noindent
\fbox{\begin{minipage}{0.95\textwidth}
\textbf{200 OK} --- Login bem-sucedido, token retornado
\end{minipage}}

\begin{minted}{json}
{
  "id": "069d510e-a565-7add-8000-65391d41016c",
  "access_token": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
  "token_type": "Bearer"
}
\end{minted}

\noindent
\fbox{\begin{minipage}{0.95\textwidth}
\textbf{401 Unauthorized} --- Credenciais inválidas
\end{minipage}}

\begin{minted}{json}
{
  "detail": "Email or password is invalid"
}
\end{minted}

\noindent
\fbox{\begin{minipage}{0.95\textwidth}
\textbf{422 Unprocessable Content} --- Formato inválido
\end{minipage}}

\begin{minted}{json}
{
  "detail": [
    {
      "type": "missing",
      "loc": ["body", "email"],
      "msg": "Field required"
    }
  ]
}
\end{minted}

\subsection{POST /api/v1/auth/logout}

\fbox{\begin{minipage}{0.95\textwidth}
\textbf{Finalidade:} Encerrar sessão do usuário

\textbf{Permissão:} Bearer Token (JWT)
\end{minipage}}

\textbf{Parâmetros:} Nenhum

\noindent\textbf{Request Body}

\begin{minted}{json}
{}
\end{minted}

\noindent\textbf{Respostas}

\noindent
\fbox{\begin{minipage}{0.95\textwidth}
\textbf{200 OK} --- Logout bem-sucedido
\end{minipage}}

\begin{minted}{json}
{
  "message": "Logout successful"
}
\end{minted}

\noindent
\fbox{\begin{minipage}{0.95\textwidth}
\textbf{401 Unauthorized} --- Token inválido ou ausente
\end{minipage}}

\begin{minted}{json}
{
  "detail": "Invalid token"
}
\end{minted}
