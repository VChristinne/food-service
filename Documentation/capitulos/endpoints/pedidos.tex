\section{Pedidos}

\subsection{POST /api/v1/orders}

\noindent\textbf{Finalidade}: Criar um novo pedido

\noindent\textbf{Permissão}: JWT | Perfil: Cliente

\noindent\textbf{Body}: Todos os campos são obrigatórios exceto \texttt{notes}, \texttt{table\_number} e \texttt{delivery\_address}

\vspace{-0.2cm}

\begin{minted}{json}
{
  "channel": "app",
  "type": "delivery",
  "items": [
    {
        "name": "Hambúrguer com Batata Frita",
        "quantity": 1
    },
    {
        "name": "Brigadeiro (6un)",
        "quantity": 1
    }
  ],
  "notes": "",
  "payment_method": "pix"
}
\end{minted}

\noindent\textbf{Status Codes}: 201, 422
