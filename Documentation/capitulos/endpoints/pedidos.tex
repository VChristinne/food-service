\section{Pedidos}

\subsection{POST /api/v1/orders/:store\_id/:costumer\_id}

\noindent\textbf{Finalidade}: Criar um novo pedido

\noindent\textbf{Permissão}: JWT | Perfil: Cliente

\noindent\textbf{Body}: Todos os campos são obrigatórios exceto \texttt{notes}, \texttt{table\_number} e \texttt{delivery\_address}

\vspace{-0.2cm}

\begin{minted}{json}
{
  "channel": "app",
  "type": "takeaway",
  "items": [
    {
      "dish_id": "{{ _.frangoassadoID }}",
      "quantity": 1
    }
  ],
  "payment_method": "pix"
}
\end{minted}

\noindent\textbf{Status Codes}: 201, 422

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{201 Created}]
\begin{minted}{json}
{
  "channel": "app",
  "points_earned": 17,

  "status": "pending",

    {
      "dish_id": "069fe03b-5379-7884-8000-4b52ad6a431f",
