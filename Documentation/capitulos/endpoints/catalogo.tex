\section{Catálogo}

\subsection{\texttt{POST /api/v1/catalogue}}

\noindent\textbf{Finalidade}: Adicionar um novo prato ao catálogo

\noindent\textbf{Permissão}: JWT | Perfil: Manager

\noindent\textbf{Body}: Todos os campos são obrigatórios

\begin{minted}{json}
{
  "name": "Hambúrguer com Batata Frita",
  "items": [
    {"ingredient_id": "id_carne_bovina", "quantity": 200.0},
    {"ingredient_id": "id_batata_frita", "quantity": 150.0}
  ],
  "price": 25.90,
  "available": true,
  "store_id": "069f3640-e1bc-7fa2-8000-a6059a7ab69a"
}
\end{minted}

\noindent\textbf{Status Codes}: 201, 403, 422

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{201 Created}]
\begin{minted}{json}
{
  "price": "25.90",
  "name": "Hambúrguer com Batata Frita",
  "store_id": "069f3640-e1bc-7fa2-8000-a6059a7ab69a",
  "updated_at": 1778085006,
  "available": true,
  "id": "069fb6c8-e763-7d3e-8000-39308786ceea",
  "created_at": 1778085006,
  "deleted_at": null
}
\end{minted}
\end{tcolorbox}

\subsection{\texttt{DELETE /api/v1/catalogue/:dish\_id}}

\noindent\textbf{Finalidade}: Remover um prato do catálogo

\noindent\textbf{Permissão}: JWT | Perfil: Manager

\noindent\textbf{Body}: Nenhum campo é necessário

\noindent\textbf{Path Parameters}: \texttt{dish\_id}

\noindent\textbf{Status Codes}: 204, 403, 404

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{204 No Content}]
Prato excluído com sucesso. Nenhum conteúdo é retornado.
\end{tcolorbox}
