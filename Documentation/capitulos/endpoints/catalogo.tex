\section{Catálogo}

\subsection{\texttt{POST /api/v1/catalogue}}

\noindent\textbf{Finalidade}: Adicionar um novo prato ao catálogo

\noindent\textbf{Permissão}: JWT | Perfil: Manager

\noindent\textbf{Body}: Todos os campos são obrigatórios

\begin{minted}{json}
{
  "name": "Frango Assado",
  "price": 15.90,
  "available": true,
  "items": [
    {"ingredient_id": "{{ _.frangoID }}", "quantity": 1000.0}
  ]
}
\end{minted}

\noindent\textbf{Status Codes}: 201, 403, 422

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{201 Created}]
\begin{minted}{json}
{
  "name": "Frango Assado",
  "price": "15.90",
  "updated_at": 1778254773,
  "store_id": "069f3640-e1bc-7fa2-8000-a6059a7ab69a",
  "created_at": 1778254773,
  "id": "069fe03b-5379-7884-8000-4b52ad6a431f",
  "deleted_at": null,
  "available": true
}
\end{minted}
\end{tcolorbox}

\subsection{\texttt{GET /api/v1/catalogue/:store_id}}

\noindent\textbf{Finalidade}: Listar todos os produtos do catálogo

\noindent\textbf{Permissão}: Público

\noindent\textbf{Body}: Nenhum campo é necessário

\noindent\textbf{Query Parameters}: \texttt{page} e \texttt{page\_size}

\noindent\textbf{Status Codes}: 200, 422

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{200 OK}]
\begin{minted}{json}
{
  "data": [
    {
      "available": true,
      "name": "Frango Assado",
      "created_at": 1778254773,
      "deleted_at": null,
      "id": "069fe03b-5379-7884-8000-4b52ad6a431f",
      "price": "15.90",
      "store_id": "069f3640-e1bc-7fa2-8000-a6059a7ab69a",
      "updated_at": 1778254773
    }
  ],
  "total": 1,
  "page": 1,
  "page_size": 10,
  "total_pages": 1
}
\end{minted}
\end{tcolorbox}

\subsection{\texttt{PATCH /api/v1/catalogue/:dish\_id}}

\noindent\textbf{Finalidade}: Atualizar os detalhes de um prato específico

\noindent\textbf{Permissão}: JWT | Perfil: Manager

\noindent\textbf{Body}: Todos os campos são opcionais

\begin{minted}{json}
{
  "price": 25.50
}
\end{minted}

\noindent\textbf{Status Codes}: 200, 403, 404, 422

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{200 OK}]
Para fins de visualização breve, é mostrado apenas os campos atualizados.
\begin{minted}{json}
{
    "id": "069fe03b-5379-7884-8000-4b52ad6a431f",
    "price": "15.50",
    "updated_at": 1778256939
}
\end{minted}
\end{tcolorbox}

\subsection{\texttt{DELETE /api/v1/catalogue/:dish\_id}}

\noindent\textbf{Finalidade}: Remover um prato do catálogo

\noindent\textbf{Permissão}: JWT | Perfil: Manager

\noindent\textbf{Body}: Nenhum campo é necessário

\noindent\textbf{Status Codes}: 204, 403, 404

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{204 No Content}]
Prato excluído com sucesso. Nenhum conteúdo é retornado.
\end{tcolorbox}
