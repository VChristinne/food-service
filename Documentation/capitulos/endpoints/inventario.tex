\section{Gestão de Inventário}

\subsection{\texttt{POST /api/v1/inventory}}

\noindent\textbf{Finalidade}: Adicionar um novo item ao inventário

\noindent\textbf{Permissão}: JWT | Perfil: Manager

\noindent\textbf{Body}: Todos os campos são obrigatórios

\vspace{-0.2cm}

\begin{minted}{json}
{
  "name": "Frango",
  "quantity": 5000,
  "unit": "g",
  "min_quantity": 2000,
  "store_id": "{{ _.storeID }}"
}
\end{minted}

\noindent\textbf{Status Codes}: 201, 401, 403, 422

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{201 Created}]
\begin{minted}{json}
{
  "unit": "g",
  "id": "069fdfb5-7daa-777a-8000-5f5d01cd4047",
  "store_id": "069f3640-e1bc-7fa2-8000-a6059a7ab69a",
  "updated_at": 1778252631,
  "name": "Frango",
  "quantity": "5000.000",
  "min_quantity": "2000.000",
  "created_at": 1778252631,
  "deleted_at": null
}
\end{minted}
\end{tcolorbox}

\subsection{\texttt{GET /api/v1/inventory/:store\_id}}

\noindent\textbf{Finalidade}: Listar itens do inventário da loja associada ao usuário autenticado

\noindent\textbf{Permissão}: JWT | Perfil: Manager

\noindent\textbf{Body}: Nenhum campo é necessário

\noindent\textbf{Status Codes}: 200, 401, 403

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{200 OK}]
Para fins de visualização breve, o item criado anteriormente está listado truncado.
\begin{minted}{json}
{
  "data": [
    {
      "unit": "g",
      "id": "069fdfb5-7daa-777a-8000-5f5d01cd4047",
      "store_id": "069f3640-e1bc-7fa2-8000-a6059a7ab69a",

      "name": "Frango",
      "quantity": "5000.000",
      "min_quantity": "2000.000",
      "created_at": 1778252631,

    }
  ],
  "total": 1,
  "page": 1,
  "page_size": 10,
  "total_pages": 1
}
\end{minted}
\end{tcolorbox}

\subsection{\texttt{DELETE /api/v1/inventory/:item\_id}}

\noindent\textbf{Finalidade}: Excluir um item específico do inventário

\noindent\textbf{Permissão}: JWT | Perfil: Manager

\noindent\textbf{Body}: Nenhum campo é necessário

\noindent\textbf{Status Codes}: 204, 401, 403, 404

\begin{tcolorbox}[colback=green!5, colframe=black, title=\textbf{204 No Content}]
Item excluído com sucesso. Nenhum conteúdo é retornado.
\end{tcolorbox}
