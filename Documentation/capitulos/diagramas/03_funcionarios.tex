\subsection{UC-03: Gestão de Funcionários}

\begin{itemize}[topsep=1pt, itemsep=0pt]
  \item[]\textbf{Descrição}: Administrador adiciona, visualiza, atualiza e remove funcionários do sistema.

  \item[]\textbf{Ator Principal}: Administrador

  \item[]\textbf{Atores Secundários}: Gerente e Funcionário

  \textbf{Pré-condições}:
  \begin{enumerate}
    \item Administrador autenticado;
    \item Loja existe no sistema (para managers e funcionários);
    \item Sistema está disponível.
  \end{enumerate}

  \textbf{Pós-condições}:
  \begin{enumerate}
    \item Funcionário criado, atualizado ou removido do sistema;
    \item Permissões aplicadas conforme o papel atribuído;
    \item Dados de funcionário refletem mudanças imediatamente;
    \item Auditoria registra todas as operações;
    \item Funcionário pode acessar sistema com email e senha.
  \end{enumerate}

  \textbf{Fluxo Principal - Criar Funcionário}:
  \begin{enumerate}
    \item Administrador acessa a interface de gerenciamento de funcionários;
    \item Administrador clica em "Adicionar Novo Funcionário";
    \item Administrador preenche os dados: nome, email, senha, telefone, CEP, complemento, loja, papel;
    \item Sistema valida os dados;
    \item Funcionário é criado no banco de dados;
    \item Sistema associa o funcionário à loja especificada;
    \item Ação é registrada na auditoria.
  \end{enumerate}

  \textbf{Fluxo Alternativo - Listar Funcionários}:
  \begin{enumerate}
    \item Administrador acessa a lista de funcionários;
    \item Sistema retorna lista paginada de todos os funcionários;
    \item Manager vê apenas funcionários de sua loja;
    \item Admin vê todos os funcionários.
  \end{enumerate}

  \textbf{Fluxo Alternativo - Visualizar Detalhes}:
  \begin{enumerate}
    \item Administrador ou Manager seleciona um funcionário;
    \item Sistema retorna todas as informações: ID, nome, email, telefone, papel, endereço completo, loja, datas;
  \end{enumerate}

  \textbf{Fluxo Alternativo - Atualizar Funcionário}:
  \begin{enumerate}
    \item Administrador ou Manager seleciona um funcionário para atualizar;
    \item Sistema valida os novos dados;
    \item Sistema verifica permissões:
    \begin{enumerate}
      \item Manager só pode atualizar funcionários da sua loja;
      \item Admin pode alterar qualquer coisa.
    \end{enumerate}
    \item Funcionário é atualizado no banco de dados;
    \item Sistema retorna os dados atualizados;
    \item Ação é registrada na auditoria com o que foi alterado.
  \end{enumerate}

  \textbf{Fluxo Alternativo - Remover Funcionário}:
  \begin{enumerate}
    \item Administrador seleciona um funcionário para remover;
    \item Administrador confirma a exclusão;
    \item Sistema realiza soft delete (marca deleted\_at com timestamp);
    \item Funcionário não pode mais fazer login;
    \item Ação é registrada na auditoria.
  \end{enumerate}

  \textbf{Regras de Negócio}:
  Consulte seção \ref{subsec:employee} para regras de negócio relacionadas à gestão de funcionários.
\end{itemize}
