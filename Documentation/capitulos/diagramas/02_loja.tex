\subsection{UC-02: Cadastro de Loja}

\begin{itemize}[topsep=1pt, itemsep=0pt]
  \item[]\textbf{Descrição}: Cadastro de loja no sistema.

  \item[]\textbf{Ator Principal}: Administrador.
  \item[]\textbf{Ator Secundário}: Gerente.

  \item[]\textbf{Sistema Externo}: API ViaCEP.

  \textbf{Pré-condições}:
  \begin{enumerate}
    \item Administrador autenticado no sistema;
    \item Loja não possui cadastro no sistema;
    \item Sistema está disponível.
  \end{enumerate}

  \textbf{Pós-condições}:
  \begin{enumerate}
    \item Loja registrada no sistema com dados válidos;
    \item Loja pode ser gerenciada e associada a funcionários pelo Gerente;
  \end{enumerate}

  \textbf{Fluxo Principal}:
  \begin{enumerate}
    \item Administrador acessa a página de cadastro de loja e preenche o formulário;
    \item Sistema valida os dados e salva a loja no banco;
    \item Sistema exibe mensagem de sucesso e redireciona para a lista de lojas.
  \end{enumerate}

  \textbf{Regras de Negócio}:
  Consulte \ref{subsec:store}.

\end{itemize}
