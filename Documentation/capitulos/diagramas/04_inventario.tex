\subsection{UC-04: Gestão de Inventário}

\begin{itemize}[topsep=1pt, itemsep=0pt]
  \item[]\textbf{Descrição}: Gerente de loja adiciona, visualiza, atualiza e remove itens do inventário.

  \item[]\textbf{Ator Principal}: Gerente.

  \textbf{Pré-condições}:
  \begin{enumerate}
    \item Gerente autenticado;
    \item Loja associada ao Gerente existe;
    \item Sistema está disponível.
  \end{enumerate}

  \textbf{Pós-condições}:
  \begin{enumerate}
    \item Item de inventário adicionado, atualizado ou removido;
    \item Inventário reflete mudanças imediatamente;
    \item Auditoria registra todas as operações;
    \item Sistema alerta quando quantidade é menor que mínimo.
  \end{enumerate}

  \textbf{Fluxo Principal - Adicionar Item}:
  \begin{enumerate}
    \item Gerente acessa a página de inventário e preenche o formulário para adicionar um novo item;
    \item Sistema valida os dados;
    \item Item é criado no inventário;
    \item Sistema retorna os detalhes do item criado;
    \item Ação é registrada na auditoria.
  \end{enumerate}

  \textbf{Fluxo Alternativo - Listar Itens}:
  \begin{enumerate}
    \item Gerente acessa o inventário;
    \item Sistema retorna lista de todos os itens do inventário da loja.
  \end{enumerate}

  \textbf{Fluxo Alternativo - Visualizar Detalhes}:
  \begin{enumerate}
    \item Gerente seleciona um item específico;
    \item Sistema retorna todas as informações.
  \end{enumerate}

  \textbf{Fluxo Alternativo - Atualizar Item}:
  \begin{enumerate}
    \item Gerente seleciona um item para atualizar;
    \item Sistema valida os novos dados;
    \item Item é atualizado no banco de dados;
    \item Sistema retorna os dados atualizados;
    \item Ação é registrada na auditoria.
  \end{enumerate}

  \textbf{Fluxo Automático - Redução ao Criar Pedido}:
  \begin{enumerate}
    \item Cliente cria um pedido com itens;
    \item Sistema identifica os ingredientes necessários para cada prato;
    \item Sistema retorna sucesso ou erro se não houver ingrediente suficiente.
  \end{enumerate}

  \textbf{Fluxo Alternativo - Remover Item}:
  \begin{enumerate}
    \item Gerente seleciona um item para remover;
    \item Gerente confirma a exclusão;
    \item Sistema remove o item do inventário;
    \item Sistema remove as associações com pratos (dish\_ingredient);
    \item Ação é registrada na auditoria.
  \end{enumerate}

  \textbf{Regras de Negócio}:
  Consulte \ref{subsec:inventory}.

\end{itemize}
