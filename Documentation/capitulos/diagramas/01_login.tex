\subsection{UC-01: Login no Sistema}

\begin{itemize}[topsep=1pt, itemsep=0pt]
  \item[]\textbf{Descrição}: Login de usuários no sistema.

  \item[]\textbf{Atores Principais}: Administrador, Cliente e Funcionário

  \textbf{Pré-condições}:
  \begin{enumerate}
    \item Usuário possui conta ativa no sistema;
    \item Sistema está disponível.
  \end{enumerate}

  \textbf{Pós-condições}:
  \begin{enumerate}
    \item Usuário autenticado no sistema;
    \item Token JWT gerado com funções apropriadas;
    \item Usuário redirecionado para a página correspondente ao seu perfil.
  \end{enumerate}

  \textbf{Fluxo Principal}:
  \begin{enumerate}
    \item Usuário acessa a página de login;
    \item Sistema exibe formulário solicitando email e senha;
    \item Usuário preenche o formulário e clica em "Entrar";
    \item Sistema valida as credenciais e autentica o usuário;
    \item Sistema inicia sessão e redireciona para a página correspondente.
  \end{enumerate}

  \textbf{Regras de Negócio}:
  Consulte seção \ref{subsec:auth} para regras de negócio relacionadas à autenticação e autorização.

 \end{itemize}
